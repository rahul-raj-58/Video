#!/usr/bin/env python3 """ metabase_video_insights.py Pulls data from your Metabase instance and assembles a real dashboard covering: - Delivery insights (turnaround, status, SLA) - Incoming video insights (volume, source, backlog) Run this YOURSELF against your live Metabase instance — it needs network access to METABASE_BASE_URL, which this chat environment doesn't have. -------------------------------------------------------------------------- SETUP -------------------------------------------------------------------------- 1. pip install requests 2. Set your credentials as environment variables (don't hardcode the key): export METABASE_API_KEY="mb_xxx..." export METABASE_BASE_URL="https://metabase.spyne.ai" export METABASE_CARD_ID="12683" Optional — add more existing Metabase questions/cards to pull in, grouped into the two sections of the dashboard: export METABASE_EXTRA_CARDS="1024:Incoming Volume by Day:incoming,1030:SLA Breaches:delivery" Format per card: "::
"
must be "incoming" or "delivery". 3. python metabase_video_insights.py - Fetches each card's results - Saves each as a CSV in ./exports/ (overwritten every run — always latest) - Prints a quick summary (row count, columns, basic stats) - Creates (or updates) a Metabase dashboard named "Delivery & Incoming Video Insights" containing all the cards, laid out in two labeled sections. - By default this now RUNS FOREVER, re-pulling every 30 minutes. Run once and exit instead with: python metabase_video_insights.py --once Change the interval with: export METABASE_REFRESH_MINUTES="15" -------------------------------------------------------------------------- NOTES -------------------------------------------------------------------------- - This script only *arranges existing questions* into a dashboard. It does not invent new SQL — Metabase card 12683 is treated as your first known question. To build out full "incoming videos" and "delivery" sections, create a couple more saved Questions in Metabase (e.g. "Videos received per day", "Avg delivery turnaround", "SLA breach count") and add their IDs via METABASE_EXTRA_CARDS above. - Metabase API keys authenticate via the `x-api-key` header. - EASIER ALTERNATIVE: you don't need this script running 24/7 just to keep the *dashboard view* fresh. Once the dashboard exists in Metabase, open it there and click the clock/refresh icon in the top right — Metabase can auto-refresh any dashboard on its own (1/5/10/15/30/60 min), pulling live query results with zero extra infrastructure. Run this script once (or on a schedule) mainly if you also want the CSV exports kept current, or want new cards auto-added to the dashboard layout. """ import os import sys import csv import json import time from datetime import datetime try: import requests except ImportError: sys.exit("Missing dependency. Run: pip install requests") BASE_URL = os.environ.get("METABASE_BASE_URL", "").rstrip("/") API_KEY = os.environ.get("METABASE_API_KEY", "") DEFAULT_CARD_ID = os.environ.get("METABASE_CARD_ID", "") EXTRA_CARDS_RAW = os.environ.get("METABASE_EXTRA_CARDS", "") DASHBOARD_NAME = "Delivery & Incoming Video Insights" EXPORT_DIR = "exports" REFRESH_MINUTES = float(os.environ.get("METABASE_REFRESH_MINUTES", "30")) HEADERS = {"x-api-key": API_KEY, "Content-Type": "application/json"} def require_config(): missing = [] if not BASE_URL: missing.append("METABASE_BASE_URL") if not API_KEY: missing.append("METABASE_API_KEY") if not DEFAULT_CARD_ID: missing.append("METABASE_CARD_ID") if missing: sys.exit(f"Missing required env vars: {', '.join(missing)}") def build_card_list(): """Returns list of dicts: {id, title, section}""" cards = [{"id": int(DEFAULT_CARD_ID), "title": "Card 12683", "section": "delivery"}] if EXTRA_CARDS_RAW.strip(): for entry in EXTRA_CARDS_RAW.split(","): entry = entry.strip() if not entry: continue parts = entry.split(":") if len(parts) != 3: print(f"Skipping malformed METABASE_EXTRA_CARDS entry: {entry}") continue cid, title, section = parts section = section.strip().lower() if section not in ("incoming", "delivery"): print(f"Skipping entry with invalid section '{section}': {entry}") continue cards.append({"id": int(cid.strip()), "title": title.strip(), "section": section}) return cards def fetch_card_data(card_id: int): """Runs the saved question and returns (columns, rows) as plain python data.""" url = f"{BASE_URL}/api/card/{card_id}/query" resp = requests.post(url, headers=HEADERS, timeout=60) resp.raise_for_status() payload = resp.json() data = payload.get("data", {}) cols = [c.get("display_name") or c.get("name") for c in data.get("cols", [])] rows = data.get("rows", []) return cols, rows def export_csv(card_id, title, cols, rows): os.makedirs(EXPORT_DIR, exist_ok=True) safe_title = "".join(c if c.isalnum() or c in " _-" else "_" for c in title).strip().replace(" ", "_") path = os.path.join(EXPORT_DIR, f"{card_id}_{safe_title}.csv") with open(path, "w", newline="", encoding="utf-8") as f: writer = csv.writer(f) writer.writerow(cols) writer.writerows(rows) return path def summarize(title, cols, rows): print(f"\n--- {title} ---") print(f"Columns: {cols}") print(f"Rows: {len(rows)}") if rows: print(f"Sample row: {rows[0]}") def get_or_create_dashboard(): # Check if a dashboard with this name already exists resp = requests.get(f"{BASE_URL}/api/dashboard", headers=HEADERS, timeout=30) resp.raise_for_status() for d in resp.json(): if d.get("name") == DASHBOARD_NAME: return d["id"] resp = requests.post( f"{BASE_URL}/api/dashboard", headers=HEADERS, json={"name": DASHBOARD_NAME, "description": "Auto-built: delivery + incoming video insights"}, timeout=30, ) resp.raise_for_status() return resp.json()["id"] def assemble_dashboard(dashboard_id, cards): """Lays out a heading + cards for each section, stacked vertically.""" dashcards = [] row_cursor = 0 section_order = ["incoming", "delivery"] grouped = {s: [c for c in cards if c["section"] == s] for s in section_order} for section in section_order: section_cards = grouped[section] if not section_cards: continue heading_text = "📥 Incoming Videos" if section == "incoming" else "🚚 Delivery" dashcards.append({ "id": -(len(dashcards) + 1), "card_id": None, "row": row_cursor, "col": 0, "size_x": 24, "size_y": 2, "visualization_settings": { "virtual_card": {"display": "heading"}, "text": f"## {heading_text}", }, }) row_cursor += 2 col_cursor = 0 for c in section_cards: dashcards.append({ "id": -(len(dashcards) + 1), "card_id": c["id"], "row": row_cursor, "col": col_cursor, "size_x": 12, "size_y": 8, }) col_cursor += 12 if col_cursor >= 24: col_cursor = 0 row_cursor += 8 row_cursor += 9 resp = requests.put( f"{BASE_URL}/api/dashboard/{dashboard_id}", headers=HEADERS, json={"dashcards": dashcards}, timeout=30, ) resp.raise_for_status() return resp.json() def run_once(cards): """Fetch fresh data for every card, re-export CSVs, and rebuild the dashboard.""" print(f"\n[{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}] Refreshing {len(cards)} card(s): {[c['title'] for c in cards]}") for c in cards: try: cols, rows = fetch_card_data(c["id"]) except requests.HTTPError as e: print(f"Failed to fetch card {c['id']} ({c['title']}): {e}") continue summarize(c["title"], cols, rows) path = export_csv(c["id"], c["title"], cols, rows) print(f"Saved -> {path}") try: dashboard_id = get_or_create_dashboard() assemble_dashboard(dashboard_id, cards) print(f"Dashboard updated: {BASE_URL}/dashboard/{dashboard_id}") except requests.HTTPError as e: print(f"Could not create/update dashboard automatically: {e}") print("Your CSV exports in ./exports/ are still available.") def main(): require_config() cards = build_card_list() run_once_only = "--once" in sys.argv run_once(cards) if run_once_only: return print(f"\nEntering refresh loop — pulling live data every {REFRESH_MINUTES:g} minutes.") print("Press Ctrl+C to stop. Use --once to run a single pull instead.") while True: try: time.sleep(REFRESH_MINUTES * 60) run_once(cards) except KeyboardInterrupt: print("\nStopped.") break except requests.RequestException as e: # Network hiccup — don't die, just wait for the next cycle. print(f"Refresh failed ({e}); will retry next cycle.") if __name__ == "__main__": main()
